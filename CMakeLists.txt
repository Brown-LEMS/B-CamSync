cmake_minimum_required(VERSION 3.18)
project(BCamSync VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BCAMSYNC_BUILD_TESTS "Build deterministic validation executables" OFF)
set(BCAMSYNC_SANITIZER "" CACHE STRING "address or thread; separate build directories")
if(BCAMSYNC_SANITIZER STREQUAL "address")
  add_compile_options(-g -O1 -fno-omit-frame-pointer -fsanitize=address,undefined -fsanitize-address-use-after-scope)
  add_link_options(-fsanitize=address,undefined)
elseif(BCAMSYNC_SANITIZER STREQUAL "thread")
  add_compile_options(-g -O1 -fno-omit-frame-pointer -fsanitize=thread)
  add_link_options(-fsanitize=thread)
elseif(NOT BCAMSYNC_SANITIZER STREQUAL "")
  message(FATAL_ERROR "BCAMSYNC_SANITIZER must be empty, address, or thread")
endif()
find_package(nlohmann_json 3.10 REQUIRED)
add_library(bcamsync SHARED src/types.cpp src/phase.cpp)
add_library(BCamSync::bcamsync ALIAS bcamsync)
target_include_directories(bcamsync PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(bcamsync PRIVATE nlohmann_json::nlohmann_json)
target_compile_options(bcamsync PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
set_target_properties(bcamsync PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
install(TARGETS bcamsync EXPORT BCamSyncTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/BCamSyncConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSync)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT BCamSyncTargets NAMESPACE BCamSync:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSync)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSync)
if(BCAMSYNC_BUILD_TESTS)
  enable_testing()
  foreach(name phase types)
    add_executable(test_${name} tests/test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE bcamsync)
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
endif()
