cmake_minimum_required(VERSION 3.18)
project(BCamSync VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(HARDWARE_DEFAULT ON)
else()
  set(HARDWARE_DEFAULT OFF)
endif()
option(BCAMSYNC_WITH_LIBCAMERA "Build managed Arducam acquisition" ${HARDWARE_DEFAULT})
option(BCAMSYNC_BUILD_PYTHON "Build Python bindings" ON)
option(BCAMSYNC_BUILD_TESTS "Build deterministic validation executables" OFF)
option(BCAMSYNC_BUILD_FUZZER "Build LLVM libFuzzer timing and message validation" OFF)
set(BCAMSYNC_SANITIZER "" CACHE STRING "address or thread; separate build directories")
if(BCAMSYNC_SANITIZER STREQUAL "address")
  add_compile_options(-g -O1 -fno-omit-frame-pointer -fsanitize=address,undefined -fsanitize-address-use-after-scope)
  add_link_options(-fsanitize=address,undefined)
elseif(BCAMSYNC_SANITIZER STREQUAL "thread")
  add_compile_options(-g -O1 -fno-omit-frame-pointer -fsanitize=thread)
  add_link_options(-fsanitize=thread)
elseif(NOT BCAMSYNC_SANITIZER STREQUAL "")
  message(FATAL_ERROR "BCAMSYNC_SANITIZER must be empty, address, or thread")
endif()
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
if(BCAMSYNC_BUILD_FUZZER)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR BCAMSYNC_SANITIZER STREQUAL "thread")
    message(FATAL_ERROR "Fuzzing requires LLVM Clang with libFuzzer, separately from TSan")
  endif()
  add_executable(fuzz_timing tests/fuzz_timing.cpp src/types.cpp src/sync_types.cpp src/phase.cpp src/wire.cpp)
  target_include_directories(fuzz_timing PRIVATE include src)
  target_link_libraries(fuzz_timing PRIVATE nlohmann_json::nlohmann_json Threads::Threads)
  target_compile_options(fuzz_timing PRIVATE -g -O1 -fno-omit-frame-pointer -fsanitize=fuzzer,address,undefined)
  target_link_options(fuzz_timing PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
set(SYNC_SOURCES src/sync_types.cpp src/phase.cpp src/wire.cpp src/process.cpp src/platform.cpp src/sync_runtime.cpp src/sync.cpp)
add_library(bcamsync SHARED ${SYNC_SOURCES} src/types.cpp src/session.cpp src/fake.cpp)
add_library(BCamSync::bcamsync ALIAS bcamsync)
target_include_directories(bcamsync PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(bcamsync PRIVATE Threads::Threads nlohmann_json::nlohmann_json)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
  target_link_libraries(bcamsync PRIVATE PkgConfig::SYSTEMD)
endif()
target_compile_options(bcamsync PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
set_target_properties(bcamsync PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
if(BCAMSYNC_WITH_LIBCAMERA)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBCAMERA REQUIRED IMPORTED_TARGET libcamera>=0.5.2)
  target_sources(bcamsync PRIVATE src/arducam.cpp)
  target_link_libraries(bcamsync PRIVATE PkgConfig::LIBCAMERA)
else()
  target_sources(bcamsync PRIVATE src/arducam_unavailable.cpp)
  message(STATUS "Managed Arducam acquisition disabled; synchronization adapters remain available")
endif()
install(TARGETS bcamsync EXPORT BCamSyncTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/BCamSyncConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSync)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT BCamSyncTargets NAMESPACE BCamSync:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSync)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSync)
add_executable(bcamsync_capture examples/capture.cpp)
target_link_libraries(bcamsync_capture PRIVATE bcamsync)
target_compile_options(bcamsync_capture PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS bcamsync_capture RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(APPLE)
  set_target_properties(bcamsync_capture PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
else()
  set_target_properties(bcamsync_capture PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
if(BCAMSYNC_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_native src/bindings.cpp src/sync_bindings.cpp)
  target_link_libraries(_native PRIVATE bcamsync)
  target_compile_options(_native PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/python/bcamsync")
  if(APPLE)
    set_target_properties(_native PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
  else()
    set_target_properties(_native PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
  endif()
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/python/bcamsync")
  configure_file(python/bcamsync/__init__.py "${CMAKE_CURRENT_BINARY_DIR}/python/bcamsync/__init__.py" COPYONLY)
  configure_file(python/bcamsync/management.py "${CMAKE_CURRENT_BINARY_DIR}/python/bcamsync/management.py" COPYONLY)
  configure_file(python/bcamsync/http_management.py "${CMAKE_CURRENT_BINARY_DIR}/python/bcamsync/http_management.py" COPYONLY)
  install(TARGETS _native LIBRARY DESTINATION bcamsync)
  install(DIRECTORY python/bcamsync/ DESTINATION bcamsync FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
endif()
if(BCAMSYNC_BUILD_TESTS)
  enable_testing()
  add_executable(test_sync_boundary tests/test_sync_boundary.cpp ${SYNC_SOURCES})
  target_include_directories(test_sync_boundary PRIVATE include src)
  target_link_libraries(test_sync_boundary PRIVATE Threads::Threads nlohmann_json::nlohmann_json)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(test_sync_boundary PRIVATE PkgConfig::SYSTEMD)
  endif()
  target_compile_options(test_sync_boundary PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  add_test(NAME sync_boundary COMMAND test_sync_boundary)
  foreach(name phase types session sync frame_queue fake gain_warmup arducam_profile node_platform)
    add_executable(test_${name} tests/test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE bcamsync Threads::Threads)
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
  target_include_directories(test_session PRIVATE src)
  target_include_directories(test_sync PRIVATE src)
  target_link_libraries(test_sync PRIVATE nlohmann_json::nlohmann_json)
  target_include_directories(test_types PRIVATE src)
  target_link_libraries(test_types PRIVATE nlohmann_json::nlohmann_json)
  target_include_directories(test_node_platform PRIVATE src)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_test(NAME private_clock COMMAND test_node_platform --private-clock)
    set_tests_properties(private_clock PROPERTIES TIMEOUT 30)
  endif()
  target_include_directories(test_gain_warmup PRIVATE src)
  target_include_directories(test_arducam_profile PRIVATE src)
  if(BCAMSYNC_BUILD_PYTHON)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/python/bcamsync_host")
    configure_file(host/bcamsync_host/__init__.py "${CMAKE_CURRENT_BINARY_DIR}/python/bcamsync_host/__init__.py" COPYONLY)
    configure_file(host/bcamsync_host/http.py "${CMAKE_CURRENT_BINARY_DIR}/python/bcamsync_host/http.py" COPYONLY)
    pybind11_add_module(_test_support src/test_support_bindings.cpp)
    target_link_libraries(_test_support PRIVATE bcamsync)
    set_target_properties(_test_support PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/python/bcamsync")
    add_test(NAME python_ownership COMMAND ${Python_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_python.py)
    set_tests_properties(python_ownership PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/python")
    add_test(NAME host_client COMMAND ${Python_EXECUTABLE} -S -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p "test_host*.py")
    set_tests_properties(host_client PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}/host")
    add_test(NAME device_workflow COMMAND ${Python_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_device_validation.py)
    set_tests_properties(device_workflow PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/python")
    add_test(NAME fake_python COMMAND ${Python_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_fake_python.py)
    set_tests_properties(fake_python PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/python" TIMEOUT 120)
    add_test(NAME external_python COMMAND ${Python_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_sync_python.py)
    set_tests_properties(external_python PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/python" TIMEOUT 120)
    add_test(NAME python_management COMMAND ${Python_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p "test_*management.py")
    set_tests_properties(python_management PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/python" TIMEOUT 120)
    add_test(NAME python_preview COMMAND ${Python_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_preview_example.py)
    set_tests_properties(python_preview PROPERTIES ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/python" TIMEOUT 120)
  endif()
endif()
