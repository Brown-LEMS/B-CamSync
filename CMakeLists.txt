cmake_minimum_required(VERSION 3.18)
project(BCamSync VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(HARDWARE_DEFAULT ON)
else()
  set(HARDWARE_DEFAULT OFF)
endif()
option(BCAMSYNC_BUILD_MANAGED "Build the optional C++ managed capture add-on" OFF)
option(BCAMSYNC_WITH_LIBCAMERA "Build managed Arducam acquisition" ${HARDWARE_DEFAULT})
option(BCAMSYNC_BUILD_PYTHON "Build Python bindings" ON)
option(BCAMSYNC_BUILD_TESTS "Build deterministic validation executables" OFF)
option(BCAMSYNC_BUILD_FUZZER "Build LLVM libFuzzer timing and message validation" OFF)
set(BCAMSYNC_SANITIZER "" CACHE STRING "address or thread; separate build directories")
if(BCAMSYNC_SANITIZER STREQUAL "address")
  add_compile_options(-g -O1 -fno-omit-frame-pointer -fsanitize=address,undefined -fsanitize-address-use-after-scope)
  add_link_options(-fsanitize=address,undefined)
elseif(BCAMSYNC_SANITIZER STREQUAL "thread")
  add_compile_options(-g -O1 -fno-omit-frame-pointer -fsanitize=thread)
  add_link_options(-fsanitize=thread)
elseif(NOT BCAMSYNC_SANITIZER STREQUAL "")
  message(FATAL_ERROR "BCAMSYNC_SANITIZER must be empty, address, or thread")
endif()
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
file(GLOB RUST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/rust/src/*.rs")
set(RUST_BUILD_ARGS build --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/rust/Cargo.toml" --release --locked)
set(RUST_BUILD_ENV "CARGO_TARGET_DIR=${CMAKE_CURRENT_BINARY_DIR}/rust")
set(RUST_TEST_ARGS test --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/rust/Cargo.toml" --locked)
set(RUST_TEST_ENV "CARGO_TARGET_DIR=${CMAKE_CURRENT_BINARY_DIR}/rust-tests")
set(RUST_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/rust/release")
if(BCAMSYNC_SANITIZER STREQUAL "thread")
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Rust/C++ ThreadSanitizer checks require Clang")
  endif()
  set(BCAMSYNC_RUST_TSAN_TOOLCHAIN "$ENV{BCAMSYNC_RUST_TSAN_TOOLCHAIN}" CACHE STRING "Pinned Rust nightly for ThreadSanitizer only")
  if(BCAMSYNC_RUST_TSAN_TOOLCHAIN STREQUAL "")
    message(FATAL_ERROR "Set BCAMSYNC_RUST_TSAN_TOOLCHAIN to an installed nightly with rust-src, or use the project development image")
  endif()
  find_program(RUSTC_EXECUTABLE rustc REQUIRED)
  execute_process(COMMAND "${RUSTC_EXECUTABLE}" "+${BCAMSYNC_RUST_TSAN_TOOLCHAIN}" -vV
    RESULT_VARIABLE RUST_TSAN_RESULT OUTPUT_VARIABLE RUST_TSAN_VERSION ERROR_VARIABLE RUST_TSAN_ERROR)
  if(NOT RUST_TSAN_RESULT EQUAL 0)
    message(FATAL_ERROR "Cannot use the ThreadSanitizer Rust toolchain: ${RUST_TSAN_ERROR}")
  endif()
  string(REGEX MATCH "host: ([^\r\n]+)" RUST_TSAN_HOST "${RUST_TSAN_VERSION}")
  set(BCAMSYNC_RUST_TSAN_TARGET "${CMAKE_MATCH_1}")
  if(BCAMSYNC_RUST_TSAN_TARGET STREQUAL "")
    message(FATAL_ERROR "Cannot identify the ThreadSanitizer Rust target")
  endif()
  set(RUST_BUILD_ARGS "+${BCAMSYNC_RUST_TSAN_TOOLCHAIN}" ${RUST_BUILD_ARGS}
    -Zbuild-std --target "${BCAMSYNC_RUST_TSAN_TARGET}")
  list(APPEND RUST_BUILD_ENV "RUSTFLAGS=-Zsanitizer=thread -Zexternal-clangrt")
  set(RUST_TEST_ARGS "+${BCAMSYNC_RUST_TSAN_TOOLCHAIN}" ${RUST_TEST_ARGS}
    --release -Zbuild-std --target "${BCAMSYNC_RUST_TSAN_TARGET}")
  set(RUST_TSAN_TEST_FLAGS "-Zsanitizer=thread -Zexternal-clangrt -Clinker=${CMAKE_CXX_COMPILER} -Clink-arg=-fsanitize=thread -Clink-arg=-lstdc++")
  list(APPEND RUST_TEST_ENV "RUSTFLAGS=${RUST_TSAN_TEST_FLAGS}" "RUSTDOCFLAGS=${RUST_TSAN_TEST_FLAGS}")
  set(RUST_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/rust/${BCAMSYNC_RUST_TSAN_TARGET}/release")
endif()
set(RUST_ARCHIVE "${RUST_OUTPUT_DIRECTORY}/${CMAKE_STATIC_LIBRARY_PREFIX}bcamsync_core${CMAKE_STATIC_LIBRARY_SUFFIX}")
add_custom_command(OUTPUT "${RUST_ARCHIVE}"
  COMMAND ${CMAKE_COMMAND} -E env ${RUST_BUILD_ENV} "${CARGO_EXECUTABLE}" ${RUST_BUILD_ARGS}
  DEPENDS ${RUST_SOURCES} rust/Cargo.toml rust/Cargo.lock
  COMMENT "Build the Rust synchronization core" VERBATIM)
add_custom_target(bcamsync_rust_build DEPENDS "${RUST_ARCHIVE}")
add_library(bcamsync_rust STATIC IMPORTED)
set_target_properties(bcamsync_rust PROPERTIES IMPORTED_LOCATION "${RUST_ARCHIVE}")
add_dependencies(bcamsync_rust bcamsync_rust_build)
if(APPLE)
  set_property(TARGET bcamsync_rust PROPERTY INTERFACE_LINK_LIBRARIES "-framework Security;-framework CoreFoundation;iconv")
else()
  set_property(TARGET bcamsync_rust PROPERTY INTERFACE_LINK_LIBRARIES "${CMAKE_DL_LIBS};m")
endif()
if(BCAMSYNC_BUILD_FUZZER)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR BCAMSYNC_SANITIZER STREQUAL "thread")
    message(FATAL_ERROR "Fuzzing requires LLVM Clang with libFuzzer, separately from TSan")
  endif()
  add_executable(fuzz_timing tests/fuzz_timing.cpp addons/managed/src/types.cpp src/sync_types.cpp src/phase.cpp src/wire.cpp)
  target_include_directories(fuzz_timing PRIVATE include src addons/managed/include addons/managed/src)
  target_link_libraries(fuzz_timing PRIVATE nlohmann_json::nlohmann_json Threads::Threads bcamsync_rust)
  target_compile_options(fuzz_timing PRIVATE -g -O1 -fno-omit-frame-pointer -fsanitize=fuzzer,address,undefined)
  target_link_options(fuzz_timing PRIVATE -fsanitize=fuzzer,address,undefined)
endif()

target_compile_options(bcamsync PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
set_target_properties(bcamsync PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
install(TARGETS bcamsync EXPORT BCamSyncTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/BCamSyncConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSync)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT BCamSyncTargets NAMESPACE BCamSync:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSync)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSync)
