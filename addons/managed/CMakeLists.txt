cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(BCamSyncManaged VERSION 0.1.0 LANGUAGES CXX)
  include(GNUInstallDirs)
  include(CMakePackageConfigHelpers)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  option(BCAMSYNC_BUILD_PYTHON "Build Python bindings" ON)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(HARDWARE_DEFAULT ON)
  else()
    set(HARDWARE_DEFAULT OFF)
  endif()
  option(BCAMSYNC_WITH_LIBCAMERA "Build managed Arducam acquisition" ${HARDWARE_DEFAULT})
  if(BCAMSYNC_BUILD_PYTHON)
    find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
    execute_process(COMMAND "${Python_EXECUTABLE}" -c "import pathlib, bcamsync; print(pathlib.Path(bcamsync.__file__).parent.parent)"
      OUTPUT_VARIABLE BCAMSYNC_PYTHON_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    list(APPEND CMAKE_PREFIX_PATH "${BCAMSYNC_PYTHON_PREFIX}")
  endif()
  find_package(BCamSync 0.1 CONFIG REQUIRED)
endif()
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
add_library(bcamsync_managed SHARED src/types.cpp src/session.cpp src/fake.cpp)
add_library(BCamSync::managed ALIAS bcamsync_managed)
set_target_properties(bcamsync_managed PROPERTIES EXPORT_NAME managed VERSION ${PROJECT_VERSION} SOVERSION 0)
target_include_directories(bcamsync_managed PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(bcamsync_managed PUBLIC BCamSync::bcamsync PRIVATE Threads::Threads nlohmann_json::nlohmann_json)
target_compile_options(bcamsync_managed PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(BCAMSYNC_WITH_LIBCAMERA)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(LIBCAMERA REQUIRED IMPORTED_TARGET libcamera>=0.5.2)
  target_sources(bcamsync_managed PRIVATE src/arducam.cpp)
  target_link_libraries(bcamsync_managed PRIVATE PkgConfig::LIBCAMERA)
else()
  target_sources(bcamsync_managed PRIVATE src/arducam_unavailable.cpp)
endif()
if(APPLE)
  set_target_properties(bcamsync_managed PROPERTIES INSTALL_RPATH "@loader_path")
else()
  set_target_properties(bcamsync_managed PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS bcamsync_managed EXPORT BCamSyncManagedTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/BCamSyncManagedConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncManagedConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSyncManaged)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BCamSyncManagedConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT BCamSyncManagedTargets NAMESPACE BCamSync:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSyncManaged)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncManagedConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/BCamSyncManagedConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BCamSyncManaged)
if(BCAMSYNC_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_managed_native src/bindings.cpp)
  set_target_properties(_managed_native PROPERTIES OUTPUT_NAME _native LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python/bcamsync_managed")
  target_link_libraries(_managed_native PRIVATE bcamsync_managed)
  target_compile_options(_managed_native PRIVATE -Wall -Wextra -Wpedantic)
  if(APPLE)
    set_target_properties(_managed_native PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
  else()
    set_target_properties(_managed_native PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
  endif()
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/python/bcamsync_managed")
  configure_file(python/bcamsync_managed/__init__.py "${CMAKE_BINARY_DIR}/python/bcamsync_managed/__init__.py" COPYONLY)
  install(TARGETS _managed_native LIBRARY DESTINATION bcamsync_managed)
  install(DIRECTORY python/bcamsync_managed/ DESTINATION bcamsync_managed FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
endif()
